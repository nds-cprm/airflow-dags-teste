  function monta_string(p_parent_table in varchar2, p_parent_join_column in varchar2, p_filter_column in varchar2, p_alternative_column in varchar2, p_child_table in varchar2, p_id in number) return varchar2 is
    v_sql varchar2(1000);
    v_delimiter varchar2(2);
    v_string varchar2(1000);
    v_result varchar2(4000);
    type result_record_type is ref cursor;
    c_results result_record_type;
  begin
    v_sql := 'select distinct t2.' || nvl(p_alternative_column, 'nome_pt') || ' from '|| p_parent_table||' t1, '|| p_child_table||' t2 where t1.'|| p_filter_column||' = '||p_id||' and t2.cod_metodo = t1.'|| p_parent_join_column||' order by 1';
	open c_results for v_sql;
    loop
		fetch c_results into v_string;
	    exit when c_results%notfound;
        v_result :=  v_result || v_delimiter || v_string;
        v_delimiter := ', ';
    end loop;
    close c_results;
    return v_result;
  end;

  function monta_materiais_analisados(p_parent_table in varchar2, p_parent_join_column in varchar2, p_filter_column in varchar2, p_alternative_column in varchar2, p_child_table in varchar2, p_id in number) return varchar2 is
    v_sql varchar2(1000);
    v_delimiter varchar2(2);
    v_string varchar2(1000);
    v_result varchar2(4000);
    type result_record_type is ref cursor;
    c_results result_record_type;
  begin
    v_sql := 'select distinct t2.' || nvl(p_alternative_column, 'nome_pt') || ' from '|| p_parent_table||' t1, '|| p_child_table||' t2 where t1.'|| p_filter_column||' = '||p_id||' and t2.codigo = t1.'|| p_parent_join_column||' and trim(t2.' || nvl(p_alternative_column, 'nome_pt')||') is not null order by 1';
	open c_results for v_sql;
    loop
		fetch c_results into v_string;
	    exit when c_results%notfound;
        v_result :=  v_result || v_delimiter || v_string;
        v_delimiter := ', ';
    end loop;
    close c_results;
    return v_result;
  end;

  procedure carrega_layer(p_reset in boolean default false) is
    v_materiais varchar2(2000);
    cursor c_dados is
      select r61802.nextval,
           g.id,
           g.amostra,
           g.toponimia,
           r.nome_pt rocha,
           m.nome_pt metodo_geoposicionamento,
           n.nivel_acesso,
           g.erro_locacional,
           monta_string('geocronologia.gr_metodo', 'cod_metodo', 'cod_cadastro_geral', 'metodo', 'bibliotecas.bb_metodos', g.id) metodos,
           g.shape,
           null materiais_analisados,
           sde.st_geometry_operators.st_x_f(g.shape),
           sde.st_geometry_operators.st_y_f(g.shape),
           'WGS84'
      from gr_cadastro_geral g, bibliotecas.bb_rocha r, bibliotecas.bb_metodo_geoposicionamento m, gr_bb_nivel_acesso n
      where g.id_rocha = r.id
      and g.id_metodo_geoposicionamento = m.id
      and g.id_nivel_acesso = n.cod_nivel_acesso;
    type layer_ttype is table of geocronologia.gr_layer%rowtype;
    layer_tab layer_ttype := layer_ttype();
  begin
    if p_reset then
        execute immediate 'truncate table geocronologia.gr_layer';
    end if;
    open c_dados;
    loop
      fetch c_dados
      bulk collect into layer_tab limit 10000;
      exit when layer_tab.count = 0;
      forall i in layer_tab.first .. layer_tab.last
        insert into geocronologia.gr_layer values layer_tab(i);
      commit;
    end loop;
    close c_dados;
    
    carrega_metodos(true);
    
    for r_feicoes in (select id from gr_layer) loop
      v_materiais := monta_materiais_analisados('geocronologia.gr_layer', 'id', 'id', 'material_analisado', 'geocronologia.gr_idade_metodos', r_feicoes.id);
      update gr_layer l
      set materiais_analisados = v_materiais
      where id = r_feicoes.id;
      commit;
    end loop;
  end;
  
  procedure carrega_layer_geoserver is
    -- Parâmetros de controle.
    l_fetch_limit constant pls_integer := 1000;  -- leitura de registros por lote.
    l_commit_batch constant pls_integer := 1000; -- commits de inserts por lote.
  
    type t_amostras is table of geocronologia.gr_layer%rowtype;
    v_amostras t_amostras;
  
    type t_sdo is table of mdsys.sdo_geometry;
    v_sdos t_sdo;
    
    l_batch_start number;
    l_batch_end number;
  begin
    loop
      -- Lê em blocos de até <l_fetch_limit> registros.
      select *
      bulk collect into v_amostras
      from geocronologia.gr_layer l
      where dbms_crypto.hash(typ => 2, src => utl_raw.cast_to_raw(l.id || '|' || l.amostra || '|' || l.rocha)) not in (select g.record_hash from geoserver.gr_layer g)
      and rownum <= l_fetch_limit;
  
      exit when v_amostras.count = 0;
  
      v_sdos := t_sdo();
      v_sdos.extend(v_amostras.count);
  
      -- Conversão SDE para SDO.
      for i in 1 .. v_amostras.count loop
        begin
          select sdo_util.from_wkbgeometry(sde.st_asbinary(v_amostras(i).shape)) into v_sdos(i) from dual;
          v_sdos(i).sdo_srid := 4326;
        exception
          when others then
            v_sdos(i) := null; -- Ignora as geometrias problemáticas.
        end;
      end loop;
  
      -- Inserção e commit em lotes de <l_commit_batch>.
      l_batch_start := 1;
      while l_batch_start <= v_amostras.count loop
        l_batch_end := least(l_batch_start + l_commit_batch - 1, v_amostras.count);
        forall i in l_batch_start .. l_batch_end save exceptions
          insert into geoserver.gr_layer
          values (
            v_amostras(i).id,
            v_amostras(i).id,
            v_amostras(i).amostra,
            v_amostras(i).toponimia,
            v_amostras(i).rocha,
            v_amostras(i).metodo_geoposicionamento,
            v_amostras(i).nivel_acesso,
            v_amostras(i).erro_locacional,
            v_amostras(i).metodos,
            v_amostras(i).materiais_analisados,
            v_amostras(i).x,
            v_amostras(i).y,
            v_amostras(i).datum,
            v_sdos(i),
            dbms_crypto.hash(typ => 2, src => utl_raw.cast_to_raw(v_amostras(i).id || '|' || v_amostras(i).amostra || '|' || v_amostras(i).rocha))
          );
        commit;
        l_batch_start := l_batch_end + 1;
      end loop;
  
      -- Limpa a memória.
      v_amostras.delete;
      v_sdos.delete;
    end loop;
  exception
    when others then
      dbms_output.put_line(sqlerrm);
      rollback;
  end;

      and iso.cod_geocron_metodo = met.cod_geocron_metodo
      and bbmet.cod_metodo = met.cod_metodo
      and iso.interpretacao = int.cod_interpretacao(+)
      and iso.cod_ref_bibliogr = ref.cod_ref_bibliogr(+)
      union all
      select bbmet.metodo metodo, cod_ia_pbevap codigo, valor_idade_agrupada idade, ref.cod_ref_bibliogr cod_ref, to_char(titulo) titulo_ref, ano ano_ref
      from gr_ia_pbevap pbe,
           gr_metodo met,
           gr_bb_interpretacao int,
           bibliotecas.bb_metodos bbmet,
           gr_ref_bibliogr ref
      where met.cod_cadastro_geral = p_cod_cadastro_geral
      and pbe.cod_geocron_metodo = met.cod_geocron_metodo
      and bbmet.cod_metodo = met.cod_metodo
      and pbe.cod_interpretacao = int.cod_interpretacao(+)
      and pbe.cod_ref_bibliogr = ref.cod_ref_bibliogr(+)

      select bbmet.metodo metodo, cod_ia_concordia codigo, intercepto_sup idade, ref.cod_ref_bibliogr cod_ref, to_char(titulo) titulo_ref, ano ano_ref
      from gr_ia_concordia con,
           gr_metodo met,
